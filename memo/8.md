
但し今回のパーサはトップレベルのフェンス構文だけを抽出する。その構造の位置を返す。一つのフェンスあたり開始行、本体、終了行の3箇所。更にメタ文字とその長さも。

尚、この規格はMarkdownのフェンスと微妙に違うため、Frameという名前にしておく。構造全体はフレーム、開始や終了を示す部分をフレームと呼称する。

```javascript
const frame = new FrameTokenizer('`', '~', '+'); // フェンスで使われうるメタ文字リスト
const tokens = frame.tokenize(text); // [{char:'`', len:3, head:{start:0, end:0, info:{start:0, end:0}}, body:{start:0, end:0}, foot:{start:0, end:0, info:{start:0, end:0}}}, ...]
```

* 改行コード: `\n`のみ（`\r\n`,`\r`は対象外）
* フレーム
	* ヘッダ（先頭）
		* フェンス（柵）
			* 行頭から任意メタ文字`` ` ``（制御文字・空白文字以外）などが3以上連続したもの
			* 終端は改行か非メタ文字が最初に登場した時
		* インフォ
			* インフォは無くても良い。任意文字列が入りうるが、ヘッダフェンスと同一文字列は不可
			* 終端は改行が最初に登場した時
	* 本文
		* 改行コードを含めた任意文字列
		* 但しヘッダフェンスやそれより長い文字列は不可（親ほど長いフェンスであることを強制する。さもなくば構文エラー）
	* フッタ（末尾）
		* フェンス（ヘッダ・フェンスと同じ。必須。存在しなければ構文エラー。但し終端は改行でなくEOFでも良い）
		* インフォ（ヘッダと同様に任意で付与できる。但し終端は改行でなくEOFでも良い）

概ね上記のようなルールである。

1. 原稿を1字ずつカーソル移動し字種を確認する
2. フレーム用メタ文字を発見したらそれが3つ以上連続していることを確認する（してなければ無視）
3. ヘッダフェンスが確定する（位置、メタ文字数、インフォ）
4. ヘッダフェンスの次の行から1字ずつカーソルに戻り、ヘッダフェンスと全く同じ行を探す（存在しなければフッタフェンス無しで構文エラー）
5. フッタフェンスが確定する（位置、メタ文字数、インフォ）
6. メタ文字種、メタ文字長、ヘッダ、ボディ、フッタの3箇所の開始と終了の添字をもたせたオブジェクトをトークンとして返す

### 修正実装

```javascript
const LF = '\n';
const MIN_FENCE_LEN = 3;

export class FrameSyntaxError extends Error {
  constructor(message, index) {
    super(`${message} (index: ${index})`);
    this.name = 'FrameSyntaxError';
    this.index = index;
  }
}

export class MetaCharSet {
  constructor(metaChars) {
    const flat = metaChars.flat();
    if (flat.length === 0) {
      throw new Error('At least one meta character is required.');
    }

    this.chars = new Set();
    for (const ch of flat) {
      if (typeof ch !== 'string' || [...ch].length !== 1) {
        throw new Error(`Invalid meta character: "${ch}"`);
      }
      if (/[\s\p{Cc}]/u.test(ch)) {
        throw new Error(`Meta character cannot be whitespace or control char: "${ch}"`);
      }
      this.chars.add(ch);
    }
  }

  has(ch) {
    return this.chars.has(ch);
  }
}

/**
 * テキスト上の現在位置（インデックス）を操作・走査する単一責任クラス。
 */
export class Cursor {
  constructor(text, pos = 0) {
    this.text = text;
    this.pos = pos;
  }

  get length() {
    return this.text.length;
  }

  get eof() {
    return this.pos >= this.text.length;
  }

  peek(offset = 0) {
    const idx = this.pos + offset;
    return idx >= 0 && idx < this.text.length ? this.text[idx] : '';
  }

  advance(count = 1) {
    this.pos = Math.min(this.text.length, this.pos + count);
    return this.pos;
  }

  moveTo(nextPos) {
    this.pos = Math.max(0, Math.min(this.text.length, nextPos));
    return this.pos;
  }

  isLineStart() {
    return this.pos === 0 || this.text[this.pos - 1] === LF;
  }

  /**
   * 現在位置から指定文字が連続する長さを数える（カーソル位置自体は進めない）。
   */
  countRepeat(ch, fromPos = this.pos) {
    let idx = fromPos;
    while (idx < this.text.length && this.text[idx] === ch) {
      idx++;
    }
    return idx - fromPos;
  }

  /**
   * 現在位置から `\n` または EOF までの終端インデックスを返す（カーソル位置自体は進めない）。
   */
  findLineEnd(fromPos = this.pos) {
    let idx = fromPos;
    while (idx < this.text.length && this.text[idx] !== LF) {
      idx++;
    }
    return idx;
  }

  /**
   * 指定インデックスの直後にある `\n` を跨いだ位置を返す。`\n` が無ければその位置を返す。
   */
  afterNewlineAt(index) {
    if (index < this.text.length && this.text[index] === LF) {
      return index + 1;
    }
    return index;
  }

  slice(start, end) {
    return this.text.slice(start, end);
  }
}

export class FrameTokenizer {
  constructor(...metaChars) {
    this.metaChars = new MetaCharSet(metaChars);
  }

  tokenize(text) {
    if (typeof text !== 'string') {
      throw new TypeError('Input text must be a string.');
    }

    const cursor = new Cursor(text, 0);
    const tokens = [];

    while (!cursor.eof) {
      if (!this._isFenceCandidateAt(cursor)) {
        cursor.advance(1);
        continue;
      }

      const token = this._parseFrame(cursor, null);
      tokens.push(token);
      cursor.moveTo(cursor.afterNewlineAt(token.foot.end));
    }

    return tokens;
  }

  _isFenceCandidateAt(cursor) {
    if (!cursor.isLineStart()) return false;
    const ch = cursor.peek(0);
    if (!this.metaChars.has(ch)) return false;
    return cursor.countRepeat(ch) >= MIN_FENCE_LEN;
  }

  _parseFrame(cursor, parentLen) {
    const head = this._parseHead(cursor, parentLen);
    const bodyStart = this._resolveBodyStart(cursor, head);
    const { bodyEnd, foot } = this._scanBodyAndFindFoot(cursor, head, bodyStart);

    return {
      char: head.char,
      len: head.len,
      head: head.range,
      body: { start: bodyStart, end: bodyEnd },
      foot
    };
  }

  _parseHead(cursor, parentLen) {
    const start = cursor.pos;
    const char = cursor.peek(0);
    const len = cursor.countRepeat(char, start);

    if (len < MIN_FENCE_LEN) {
      throw new FrameSyntaxError(`Fence length must be at least ${MIN_FENCE_LEN}`, start);
    }
    if (parentLen !== null && len >= parentLen) {
      throw new FrameSyntaxError(
        `Nested fence length (${len}) must be shorter than parent fence length (${parentLen})`,
        start
      );
    }

    const fenceEnd = start + len;
    const info = this._parseLineInfo(cursor, fenceEnd, char.repeat(len), 'Header');

    return {
      char,
      len,
      range: { start, end: info.end, info }
    };
  }

  _parseLineInfo(cursor, infoStart, fenceStr, label) {
    const end = cursor.findLineEnd(infoStart);
    if (cursor.slice(infoStart, end) === fenceStr) {
      throw new FrameSyntaxError(`${label} info cannot be identical to fence`, infoStart);
    }
    return { start: infoStart, end };
  }

  _resolveBodyStart(cursor, head) {
    const nextPos = cursor.afterNewlineAt(head.range.end);
    if (nextPos === head.range.end && cursor.eof) {
      throw new FrameSyntaxError('Missing footer fence (reached EOF in header)', head.range.start);
    }
    return nextPos;
  }

  _scanBodyAndFindFoot(cursor, head, bodyStart) {
    cursor.moveTo(bodyStart);
    const fenceStr = head.char.repeat(head.len);

    while (!cursor.eof) {
      if (!cursor.isLineStart()) {
        cursor.advance(1);
        continue;
      }

      const lineChar = cursor.peek(0);
      if (!this.metaChars.has(lineChar)) {
        cursor.advance(1);
        continue;
      }

      const lineStart = cursor.pos;
      const runLen = cursor.countRepeat(lineChar, lineStart);

      if (lineChar === head.char && runLen === head.len) {
        const foot = this._parseFoot(cursor, lineStart, runLen, fenceStr);
        return { bodyEnd: lineStart, foot };
      }

      if (lineChar === head.char && runLen > head.len) {
        throw new FrameSyntaxError(
          `Fence length (${runLen}) exceeds header fence length (${head.len})`,
          lineStart
        );
      }

      if (runLen >= MIN_FENCE_LEN) {
        const child = this._parseFrame(cursor, head.len);
        cursor.moveTo(cursor.afterNewlineAt(child.foot.end));
        continue;
      }

      cursor.advance(1);
    }

    throw new FrameSyntaxError('Missing matching footer fence', head.range.start);
  }

  _parseFoot(cursor, start, len, fenceStr) {
    const info = this._parseLineInfo(cursor, start + len, fenceStr, 'Footer');
    return { start, end: info.end, info };
  }
}

```

    }

```javascript
  }

    }
  }
```

``````markdown
  #lineStarts;

  constructor(text) {
    if (!(text instanceof Text)) {
      throw new TypeError('text must be an instance of Text.');
    }
    this.#text = text;
    this.#lineStarts = this.#buildLineStarts(text.string);
  }

  get segmenter() { return this.#text.segmenter; }

  indexToRowCol(index) {
    const str = this.#text.string;
    const clamped = Math.max(0, Math.min(str.length, index));
    const rowIndex = this.#findRowIndex(clamped);
    const lineStart = this.#lineStarts[rowIndex];
    const lineSlice = str.slice(lineStart, clamped);

    let col = 1;
    for (const _ of this.segmenter.segment(lineSlice)) {
      col++;
    }

    return { row: rowIndex + 1, col };
  }

  rowColToIndex(row, col) {
    const str = this.#text.string;
    const rowIndex = Math.max(0, Math.min(this.#lineStarts.length - 1, row - 1));
    const lineStart = this.#lineStarts[rowIndex];
    const lineEnd = rowIndex + 1 < this.#lineStarts.length
      ? this.#lineStarts[rowIndex + 1] - 1
      : str.length;

    const targetColOffset = Math.max(0, col - 1);
    if (targetColOffset === 0) {
      return lineStart;
    }

    const lineSlice = str.slice(lineStart, lineEnd);
    let count = 0;
    let offset = 0;

    for (const part of this.segmenter.segment(lineSlice)) {
      if (count >= targetColOffset) {
        break;
      }
      offset += part.segment.length;
      count++;
    }

    return Math.min(lineEnd, lineStart + offset);
  }

  #buildLineStarts(str) {
    const starts = [0];
    const lf = this.#text.code;
    for (let i = 0; i < str.length; i++) {
      if (str[i] === lf) {
        starts.push(i + 1);
      }
    }
    return starts;
  }

  #findRowIndex(offset) {
    let low = 0;
    let high = this.#lineStarts.length - 1;

    while (low <= high) {
      const mid = (low + high) >> 1;
      if (this.#lineStarts[mid] <= offset) {
        low = mid + 1;
      } else {
        high = mid - 1;
      }
    }

    return Math.max(0, high);
  }
``````

````javascript
export class TextDiffError extends Error {
  constructor(message = 'Text was modified during newline normalization.') {
    super(message);
    this.name = 'TextDiffError';
  }
}

export class Text {
  static #LF = '\n';
  static #CRLF = /\r\n/g;
  static #defSeg = null;

  static get LF() { return Text.#LF; }

  static get defaultSegmenter() {
    if (!Text.#defSeg) { Text.#defSeg = new Intl.Segmenter(undefined, { granularity: 'grapheme' }); }
    return Text.#defSeg;
  }

  static #validateSegmenter(segmenter) {
    if (segmenter === null || segmenter === undefined) { return Text.defaultSegmenter; }
    if (!(segmenter instanceof Intl.Segmenter)) { throw new TypeError('segmenter must be an Intl.Segmenter instance or null.'); }
    const { granularity } = segmenter.resolvedOptions();
    if (granularity !== 'grapheme') { throw new Error(`Invalid segmenter granularity "${granularity}": expected "grapheme".`); }
    return segmenter;
  }

  static normalize(text) {
    if (typeof text !== 'string') { throw new TypeError('Input text must be a string.'); }
    return text.replace(Text.#CRLF, Text.#LF);
  }

  #string;
  #isDiff;
  #segmenter;

  constructor(text, segmenter = null) {
    if (typeof text !== 'string') { throw new TypeError('Text must be a string.'); }
    this.#segmenter = Text.#validateSegmenter(segmenter);
    this.#string = Text.normalize(text);
    this.#isDiff = text !== this.#string;
  }

  get string() { return this.#string; }
  get LF() { return Text.#LF; }
  get isDiff() { return this.#isDiff; }
  get segmenter() { return this.#segmenter; }

  valid() {
    if (this.#isDiff) { throw new TextDiffError(); }
  }
}
````
